#include <Arduino.h>
#include <Wire.h>

namespace {
constexpr int SDA_PIN = 21;
constexpr int SCL_PIN = 22;
constexpr uint32_t SCAN_INTERVAL_MS = 3000;
uint32_t lastScan = 0;
bool firstScan = true;
bool busReady = false;

void scanBus() {
    unsigned devices = 0;
    unsigned errors = 0;
    bool oledAddress = false;
    bool imuAddress = false;
    bool tofAddress = false;
    Serial.printf("\n[%lu ms] Scanning SDA=21 SCL=22 at 100 kHz\n",
                  static_cast<unsigned long>(millis()));
    for (uint8_t address = 1; address < 127; ++address) {
        Wire.beginTransmission(address);
        const uint8_t result = Wire.endTransmission();
        if (result == 0) {
            ++devices;
            Serial.printf("  ACK 0x%02X\n", address);
            oledAddress |= address == 0x3C || address == 0x3D;
            imuAddress |= address == 0x68 || address == 0x69;
            tofAddress |= address == 0x29;
        } else if (result != 2) { // Address NACK normally means no device.
            ++errors;
            Serial.printf("  BUS ERROR at 0x%02X: Wire code %u\n", address, result);
        }
        yield();
    }
    Serial.printf("Responding addresses: %u; bus errors: %u\n", devices, errors);
    Serial.printf("OLED candidate 0x3C/0x3D: %s\n", oledAddress ? "ACK" : "NO ACK");
    Serial.printf("IMU candidate 0x68/0x69: %s\n", imuAddress ? "ACK" : "NO ACK");
    Serial.printf("ToF candidate 0x29: %s\n", tofAddress ? "ACK" : "NO ACK");
    Serial.println("Addresses do not identify sensor models or OLED controllers.");
    Serial.println("Scanner only: OLED drawing and servo output are NOT enabled.");
    if (errors) Serial.println("Bus errors detected: do not interpret missing ACKs as absent modules.");
}
}

void setup() {
    Serial.begin(115200);
    busReady = Wire.begin(SDA_PIN, SCL_PIN, 100000);
    Wire.setTimeOut(25);
    Serial.println("\nBeyLit M1 I2C diagnostics - ADVISORY ONLY, NO AUTOMATIC BRAKING");
    if (!busReady) Serial.println("I2C initialization failed. Check board and pin configuration.");
}

void loop() {
    if (!busReady) {
        yield();
        return;
    }
    const uint32_t now = millis();
    if (firstScan || static_cast<uint32_t>(now - lastScan) >= SCAN_INTERVAL_MS) {
        firstScan = false;
        scanBus();
        lastScan = millis();
    }
    yield();
}
