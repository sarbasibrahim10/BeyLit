#include <Arduino.h>
#include <Wire.h>
#include <Adafruit_SH110X.h>
#include <cmath>
#include <cstring>
#include <ESP32Servo.h>

// Bounded one-axis compensation diagnostic; mechanical axis and sign need verification.
constexpr uint32_t SAMPLE_MS = 20, MAX_GAP_US = 250000;
constexpr float GRAVITY = 9.80665f, RAD_TO_DEGREES = 57.2957795f;
constexpr unsigned CAL_SAMPLES = 150, MAX_CAL_ATTEMPTS = 3;
Adafruit_SH1106G oled(128, 64, &Wire, -1, 100000, 100000);
enum class State { STARTUP, CALIBRATING, READY, ERROR };
State state = State::STARTUP;
bool busReady = false, oledReady = false, imuReady = false, orientationValid = false;
uint8_t imuAddress = 0, oledAddress = 0;
const char *fault = "NONE";
float pitch = 0, roll = 0, pitchReference = 0, rollReference = 0;
float gyroBias[3] = {}, gyroSum[3] = {};
float sumPitch = 0, sumRoll = 0, firstPitch = 0, firstRoll = 0;
float accelerationG = 0;
unsigned samples = 0, attempt = 0;
uint32_t calStarted = 0, calDeadline = 0, lastSample = 0, lastMicros = 0, lastLog = 0;
char command[32]; size_t commandLength = 0; bool overflow = false;

// Bench test only: approximate gain; not a calibrated physical servo angle.
Servo stabilizer;
bool compensation = false, useRoll = false, servoTest = false; // boot default PITCH
int direction = -1, pulseUs = 1500, targetUs = 1500, testPhase = 0;
uint32_t lastServo = 0, compensationStarted = 0, lastTrace = 0;
// Slew limit by elapsed time so OLED/serial stalls do not slow the servo; full 100 us span in ~0.25 s.
constexpr float SLEW_US_PER_MS = 0.4f;
constexpr uint32_t MAX_SLEW_GAP_MS = 100;
// Brief acceleration spikes (hand bumps) hold the last command; only a sustained excursion stops.
constexpr uint32_t ACCEL_HOLD_MS = 500;
uint32_t accelBadSince = 0; // 0 = acceleration currently inside gate
float selectedTilt() { return useRoll ? roll - rollReference : pitch - pitchReference; }
// Stops PWM only (no power disconnect, no centering); reason is printed when a test was running.
void stopCompensation(const char *reason = nullptr) {
    const bool wasRunning = compensation || servoTest;
    compensation = servoTest = false;
    if (stabilizer.attached()) stabilizer.detach();
    pinMode(18, OUTPUT); digitalWrite(18, LOW);
    if (wasRunning) Serial.printf("SERVO SIGNAL OFF t=%lu reason=%s last_command_us=%d\n",
        static_cast<unsigned long>(millis()), reason ? reason : "COMMAND", pulseUs);
}
bool attachServo() {
    stabilizer.setPeriodHertz(50);
    stabilizer.attach(18, 1450, 1550);
    if (!stabilizer.attached()) { Serial.println("SERVO ATTACH FAILED"); return false; }
    pulseUs = targetUs = 1500; stabilizer.writeMicroseconds(pulseUs);
    lastServo = lastTrace = millis();
    return true;
}
void startCompensation() {
    if (servoTest) { Serial.println("COMP refused: SERVO TEST running."); return; }
    if (state != State::READY || !orientationValid ||
        static_cast<uint32_t>(micros() - lastMicros) > MAX_GAP_US ||
        accelerationG < 0.85f || accelerationG > 1.15f) {
        Serial.println("COMP refused: calibrate and hold MPU still first."); return;
    }
    const float tilt = selectedTilt();
    if (!isfinite(tilt) || fabsf(tilt) > 30.0f) {
        Serial.printf("COMP refused: %s tilt %.1f deg beyond 30; return to reference pose.\n", useRoll ? "ROLL" : "PITCH", tilt);
        return;
    }
    if (!attachServo()) return;
    compensation = true; compensationStarted = millis(); accelBadSince = 0;
    Serial.printf("COMP ON t=%lu axis=%s sign=%d start_tilt=%.2f: 60-second test, 1450..1550 us.\n",
        static_cast<unsigned long>(compensationStarted), useRoll ? "ROLL" : "PITCH", direction, tilt);
}
// Same bounded sweep as M5 backup: 1500 -> 1450 -> 1550 -> 1500, 1 us per 40 ms, 10 s limit.
void startServoTest() {
    if (compensation) { Serial.println("SERVO TEST refused: COMP running."); return; }
    if (servoTest || !attachServo()) return;
    servoTest = true; testPhase = 0; compensationStarted = millis();
    Serial.println("SERVO TEST: neutral then 1450 -> 1550 -> 1500 us; commanded position only.");
}
void traceServo(float tilt) {
    if (static_cast<uint32_t>(millis() - lastTrace) < 100) return;
    lastTrace = millis();
    Serial.printf("TRACE t=%lu mode=%s axis=%s tilt=%.2f target_us=%d command_us=%d attached=%d accel_g=%.3f\n",
        static_cast<unsigned long>(lastTrace), servoTest ? "TEST" : "COMP", useRoll ? "ROLL" : "PITCH",
        tilt, targetUs, pulseUs, stabilizer.attached(), accelerationG);
}
void updateServoTest() {
    const uint32_t now = millis();
    if (static_cast<uint32_t>(now - compensationStarted) >= 10000) { stopCompensation("TEST 10 s LIMIT"); return; }
    if (static_cast<uint32_t>(now - compensationStarted) < 1000 ||
        static_cast<uint32_t>(now - lastServo) < 40) return;
    lastServo = now;
    const int targets[] = {1450, 1550, 1500};
    targetUs = targets[testPhase];
    pulseUs += constrain(targetUs - pulseUs, -1, 1);
    stabilizer.writeMicroseconds(pulseUs);
    traceServo(orientationValid ? selectedTilt() : NAN);
    if (pulseUs == targetUs && ++testPhase == 3) stopCompensation("TEST COMPLETE");
}
void updateCompensation() {
    if (servoTest) { updateServoTest(); return; }
    if (!compensation) return;
    char reason[64];
    if (state != State::READY || !orientationValid) { stopCompensation("IMU NOT READY"); return; }
    if (static_cast<uint32_t>(micros() - lastMicros) > MAX_GAP_US) { stopCompensation("IMU DATA STALE"); return; }
    if (accelerationG < 0.85f || accelerationG > 1.15f) {
        if (!accelBadSince) accelBadSince = millis() | 1;
        if (static_cast<uint32_t>(millis() - accelBadSince) < ACCEL_HOLD_MS) return; // hold position
        snprintf(reason, sizeof(reason), "ACCEL %.3f g OUTSIDE 0.85..1.15 FOR 0.5 s", accelerationG);
        stopCompensation(reason); return;
    }
    accelBadSince = 0;
    if (static_cast<uint32_t>(millis() - compensationStarted) >= 60000) { stopCompensation("60 s TIMEOUT"); return; }
    const uint32_t now = millis(), sinceServo = static_cast<uint32_t>(now - lastServo);
    if (sinceServo < 20) return;
    lastServo = now;
    float tilt = selectedTilt();
    if (!isfinite(tilt) || fabsf(tilt) > 30.0f) {
        snprintf(reason, sizeof(reason), "TILT %.1f deg BEYOND 30", tilt);
        stopCompensation(reason); return;
    }
    traceServo(tilt);
    if (fabsf(tilt) < 0.5f) tilt = 0;
    targetUs = constrain(1500 + static_cast<int>(lroundf(direction * tilt * 10.0f)), 1450, 1550);
    const int maxStep = static_cast<int>(lroundf(min(sinceServo, MAX_SLEW_GAP_MS) * SLEW_US_PER_MS));
    pulseUs += constrain(targetUs - pulseUs, -maxStep, maxStep);
    stabilizer.writeMicroseconds(pulseUs);
}

bool readRegisters(uint8_t address, uint8_t reg, uint8_t *data, uint8_t count) {
    Wire.beginTransmission(address); Wire.write(reg);
    if (Wire.endTransmission(false) != 0) return false;
    if (Wire.requestFrom(address, count) != count) return false;
    for (uint8_t i = 0; i < count; ++i) data[i] = Wire.read();
    return true;
}
bool responds(uint8_t address) {
    if (!busReady) return false;
    Wire.beginTransmission(address); return Wire.endTransmission() == 0;
}
const char *stateName() {
    switch (state) {
        case State::STARTUP: return "STARTUP";
        case State::CALIBRATING: return "CALIBRATING";
        case State::READY: return "READY";
        default: return "SENSOR_ERROR";
    }
}
void fail(const char *reason) {
    stopCompensation(reason);
    state = State::ERROR; orientationValid = false; fault = reason;
    Serial.printf("IMU ERROR: %s; servo OFF, compensation OFF\n", reason);
}
void beginAttempt() {
    ++attempt; samples = 0; sumPitch = sumRoll = 0;
    for (unsigned i = 0; i < 3; ++i) gyroSum[i] = 0;
    calStarted = millis();
    Serial.printf("HOLD STILL: calibration attempt %u/%u\n", attempt, MAX_CAL_ATTEMPTS);
}
void startCalibration() {
    stopCompensation("CALIBRATE");
    if (!imuReady) { fail("IMU INIT FAILED"); return; }
    state = State::CALIBRATING; orientationValid = false; fault = "NONE";
    attempt = 0; calDeadline = millis(); beginAttempt();
}
void rejectMovement() {
    if (attempt >= MAX_CAL_ATTEMPTS) { fail("CALIBRATION MOTION"); return; }
    beginAttempt();
}
// MPU6500 diagnostic: checked 14-byte burst, +/-2g and +/-250 deg/s scales.
bool readSample(float a[3], float g[3]) {
    uint8_t bytes[14];
    if (!readRegisters(imuAddress, 0x3B, bytes, sizeof(bytes))) return false;
    auto signedWord = [&](unsigned i) {
        const uint16_t value = (static_cast<uint16_t>(bytes[i]) << 8) | bytes[i + 1];
        return value < 32768 ? static_cast<int32_t>(value) : static_cast<int32_t>(value) - 65536;
    };
    for (unsigned i = 0; i < 3; ++i) {
        a[i] = signedWord(i * 2) * GRAVITY / 16384.0f; // +/-2g -> m/s^2
        g[i] = signedWord(8 + i * 2) / 131.0f / RAD_TO_DEGREES; // +/-250 dps -> rad/s
    }
    return true;
}
void updateImu() {
    if (!imuReady || state == State::ERROR) return;
    const uint32_t now = millis();
    if (static_cast<uint32_t>(now - lastSample) < SAMPLE_MS) return;
    lastSample = now;
    float a[3], g[3];
    if (!readSample(a, g)) { imuReady = false; fail("I2C READ FAILED"); return; }
    const uint32_t tick = micros();
    const uint32_t elapsed = static_cast<uint32_t>(tick - lastMicros); lastMicros = tick;
    const float norm = sqrtf(a[0]*a[0] + a[1]*a[1] + a[2]*a[2]);
    accelerationG = norm / GRAVITY;
    if (norm < 0.1f) { fail("INVALID ACCELERATION"); return; }
    const float accPitch = atan2f(-a[0], sqrtf(a[1]*a[1] + a[2]*a[2])) * RAD_TO_DEGREES;
    const float accRoll = atan2f(a[1], a[2]) * RAD_TO_DEGREES;
    if (state == State::CALIBRATING) {
        if (static_cast<uint32_t>(now - calDeadline) > 15000) { fail("CALIBRATION TIMEOUT"); return; }
        if (static_cast<uint32_t>(now - calStarted) < 1000) return; // settle before collecting
        const float speed = sqrtf(g[0]*g[0] + g[1]*g[1] + g[2]*g[2]);
        if (fabsf(accPitch) > 60 || fabsf(accRoll) > 60) {
            Serial.printf("CAL pose unsuitable: pitch=%.1f roll=%.1f deg\n", accPitch, accRoll);
            fail("REST BOARD FLAT"); return;
        }
        if (accelerationG < 0.85f || accelerationG > 1.15f || speed > 0.15f) {
            Serial.printf("CAL motion check: accel_g=%.3f gyro_radps=%.3f\n", accelerationG, speed);
            rejectMovement(); return;
        }
        if (!samples) { firstPitch = accPitch; firstRoll = accRoll; }
        if (fabsf(accPitch - firstPitch) > 2 || fabsf(accRoll - firstRoll) > 2) { rejectMovement(); return; }
        sumPitch += accPitch; sumRoll += accRoll;
        for (unsigned i = 0; i < 3; ++i) gyroSum[i] += g[i];
        if (++samples >= CAL_SAMPLES) {
            pitch = pitchReference = sumPitch / samples;
            roll = rollReference = sumRoll / samples;
            for (unsigned i = 0; i < 3; ++i) gyroBias[i] = gyroSum[i] / samples;
            state = State::READY; orientationValid = true; fault = "NONE";
            Serial.printf("CALIBRATION OK reference pitch=%.2f roll=%.2f deg; gyro bias rad/s=%.5f,%.5f,%.5f\n",
                pitchReference, rollReference, gyroBias[0], gyroBias[1], gyroBias[2]);
        }
        return;
    }
    if (elapsed > MAX_GAP_US) { fail("SAMPLE TIME GAP"); return; }
    const float dt = elapsed / 1000000.0f;
    // Simple small-angle complementary filter; board X gyro -> roll, Y -> pitch.
    pitch += (g[1] - gyroBias[1]) * RAD_TO_DEGREES * dt;
    roll += (g[0] - gyroBias[0]) * RAD_TO_DEGREES * dt;
    if (accelerationG >= 0.85f && accelerationG <= 1.15f) {
        const float alpha = 0.5f / (0.5f + dt);
        pitch = alpha * pitch + (1 - alpha) * accPitch;
        roll = alpha * roll + (1 - alpha) * accRoll;
        fault = "NONE";
    } else fault = "ACCEL CORRECTION OFF";
    orientationValid = true;
}
void draw() {
    if (!oledReady) return;
    if (!responds(oledAddress)) { oledReady = false; Serial.println("OLED ERROR; display unavailable"); return; }
    oled.clearDisplay(); oled.setTextColor(SH110X_WHITE); oled.setTextSize(1);
    oled.setTextWrap(false); oled.setCursor(0, 0); oled.println("BeyLit TILT TEST");
    oled.setCursor(0, 12);
    if (state == State::CALIBRATING) {
        oled.println("HOLD STILL"); oled.setCursor(0, 24);
        oled.printf("Samples: %u/%u", samples, CAL_SAMPLES);
    } else if (state == State::READY && orientationValid) {
        oled.printf("Pitch: %+.1f deg", pitch - pitchReference);
        oled.setCursor(0, 24); oled.printf("Roll:  %+.1f deg", roll - rollReference);
    } else { oled.println("IMU ERROR"); oled.setCursor(0, 24); oled.println(fault); }
    oled.setCursor(0, 36); oled.printf("%s %s %s %d", servoTest ? "TEST" : "COMP", compensation || servoTest ? "ON" : "OFF", useRoll ? "ROLL" : "PITCH", pulseUs);
    oled.setCursor(0, 46); oled.println("ADVISORY ONLY");
    oled.setCursor(0, 56); oled.println("NO AUTOMATIC BRAKING");
    oled.display();
}
void status() {
    Serial.printf("COMP=%s TEST=%s axis=%s sign=%d command_us=%d (not measured) ", compensation ? "ON" : "OFF", servoTest ? "ON" : "OFF", useRoll ? "ROLL" : "PITCH", direction, pulseUs);
    Serial.printf("t=%lu state=%s pitch=", static_cast<unsigned long>(millis()), stateName());
    if (orientationValid) Serial.printf("%.2f roll=%.2f", pitch-pitchReference, roll-rollReference);
    else Serial.print("null roll=null");
    Serial.printf(" accel_g=%.3f samples=%u fault=%s OLED=%s\n",
        accelerationG, samples, fault, oledReady ? "initialized" : "ERROR");
}
void inspectRaw() {
    float a[3], g[3];
    if (!imuReady || !readSample(a, g)) { Serial.println("RAW READ ERROR"); return; }
    Serial.printf("RAW accel_mps2=%.3f,%.3f,%.3f gyro_radps=%.4f,%.4f,%.4f\n",
        a[0],a[1],a[2],g[0],g[1],g[2]);
}
void inspectImu() {
    for (uint8_t address : {static_cast<uint8_t>(0x68), static_cast<uint8_t>(0x69)}) {
        uint8_t id = 0;
        bool ack = responds(address);
        bool ok = ack && readRegisters(address, 0x75, &id, 1);
        Serial.printf("IMU CHECK address=0x%02X ACK=%u id_read=%u WHO_AM_I=0x%02X\n", address, ack, ok, id);
    }
}
void commands() {
    for (unsigned i = 0; i < 64 && Serial.available(); ++i) {
        char c = Serial.read();
        if (c == '\n' || c == '\r') {
            command[commandLength] = 0;
            if (overflow) Serial.println("Command discarded: too long");
            else if (!strcmp(command, "CALIBRATE")) startCalibration();
            else if (!strcmp(command, "STATUS")) status();
            else if (!strcmp(command, "IMU CHECK")) inspectImu();
            else if (!strcmp(command, "RAW")) inspectRaw();
            else if (!strcmp(command, "COMP ON")) startCompensation();
            else if (!strcmp(command, "SERVO TEST")) startServoTest();
            else if (!strcmp(command, "COMP OFF") || !strcmp(command, "SERVO OFF")) stopCompensation();
            else if (!strcmp(command, "AXIS ROLL")) { stopCompensation(); useRoll = true; }
            else if (!strcmp(command, "AXIS PITCH")) { stopCompensation(); useRoll = false; }
            else if (!strcmp(command, "REVERSE")) { stopCompensation(); direction = -direction; }
            else if (commandLength) Serial.println("Commands: STATUS | CALIBRATE | SERVO TEST | COMP ON/OFF | AXIS ROLL/PITCH | REVERSE");
            commandLength = 0; overflow = false;
        } else if (commandLength < sizeof(command)-1 && !overflow) command[commandLength++] = c;
        else overflow = true;
    }
}
void setup() {
    Serial.begin(115200);
    pinMode(18, OUTPUT); digitalWrite(18, LOW); // Servo off until COMP ON.
    pinMode(19, OUTPUT); digitalWrite(19, LOW);
    pinMode(27, OUTPUT); digitalWrite(27, LOW); // second servo removed; keep pin quiet
    busReady = Wire.begin(21, 22, 100000); Wire.setTimeOut(25);
    if (!busReady) { fail("I2C INIT FAILED"); return; }
    oledAddress = responds(0x3C) ? 0x3C : responds(0x3D) ? 0x3D : 0;
    if (oledAddress) oledReady = oled.begin(oledAddress, false);
    if (!oledReady) Serial.println("OLED ERROR: user-facing demo unavailable");
    draw();
    imuAddress = responds(0x68) ? 0x68 : responds(0x69) ? 0x69 : 0;
    inspectImu();
    if (!imuAddress) { fail("IMU NO ACK"); return; }
    uint8_t identity = 0;
    if (!readRegisters(imuAddress, 0x75, &identity, 1) || identity != 0x70) {
        fail("UNSUPPORTED IMU ID"); return;
    }
    // MPU6500-specific register setup, with checked writes and readback.
    // Register values follow TDK RM-MPU-6500A-00; do not bypass identity checks.
    const uint8_t registers[] = {0x6B, 0x6C, 0x1A, 0x1B, 0x1C, 0x1D, 0x19};
    const uint8_t values[] =    {0x01, 0x00, 0x03, 0x00, 0x00, 0x03, 0x09};
    for (unsigned i = 0; i < sizeof(registers); ++i) {
        Wire.beginTransmission(imuAddress);
        Wire.write(registers[i]); Wire.write(values[i]);
        if (Wire.endTransmission() != 0) { fail("IMU WRITE ERROR"); return; }
        uint8_t actual = 0;
        if (!readRegisters(imuAddress, registers[i], &actual, 1) || actual != values[i]) {
            fail("IMU CONFIG ERROR"); return;
        }
    }
    imuReady = true; lastMicros = micros();
    Serial.printf("MPU6500 diagnostic initialized at 0x%02X; HOLD STILL.\n", imuAddress);
    startCalibration(); draw();
}
void loop() {
    commands(); updateImu(); updateCompensation();
    if (static_cast<uint32_t>(millis()-lastLog) >= 500) {
        lastLog = millis(); status(); draw();
    }
    yield();
}
